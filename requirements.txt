# The foundation model. Its licence is non-commercial and it is not
# redistributed with this package: install it and let it fetch its own
# checkpoints, then point the loaders at them.
#
#   pip install "tabpfn>=2.0"
#   export SPEC_TABPFN_MODELS=/path/to/tabpfn-checkpoints
#
# code/common/paths.py lists the three checkpoint file names the loaders expect.
# The generations are the ones the paper calls TabPFN v2.5, v3 and v3.5; not
# every result family uses all three, and each result file names the generation
# behind it. Without the weights the trust-layer and representation results
# cannot be rerun, but every number in them is present in results/ as shipped.
#
# Only release 9.0.0 was verified here. The code constructs its estimators with
# a `model_path` argument; if an older release rejects that argument, install a
# current one.
